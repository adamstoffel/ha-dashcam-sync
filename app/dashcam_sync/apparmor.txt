#include <tunables/global>

# Custom AppArmor profile for Dashcam Sync (+1 on the security rating).
# Roll out in complain mode through the beta, then enforce.
# The Supervisor rewrites the profile name to the app's slug when it loads this file.
profile dashcam_sync flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities and signals
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /etc/s6-overlay/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Options, catalogue and clips
  /data/** rwk,
  /media/** rwk,
  /share/** rwk,

  # Networking: the app talks to cameras, the Supervisor and Home Assistant Core
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network netlink raw,

  # Python and the libraries it loads
  /usr/bin/python3* ix,
  /opt/venv/** rix,
  /usr/lib/python3*/** r,
  /usr/lib/** mr,
  /lib/** mr,

  # ffmpeg parses camera-made files: run it and nothing else
  /usr/bin/ffmpeg ix,
  /usr/bin/ffprobe ix,
  /usr/bin/ping Ux,

  # Mount health check and runtime information
  /proc/self/mountinfo r,
  /proc/self/** r,
  /proc/sys/net/core/somaxconn r,
  /sys/devices/system/cpu/** r,
  /etc/ssl/** r,
  /etc/resolv.conf r,
  /etc/hosts r,
  /etc/nsswitch.conf r,

  # Deny everything else explicitly
  deny /root/** rwx,
  deny /home/** rwx,
  deny /etc/shadow rwx,
  deny /boot/** rwx,
  deny /sys/kernel/** rwx,
}
